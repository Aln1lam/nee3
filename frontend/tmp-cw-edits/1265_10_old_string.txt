    const DEFAULT_CATEGORIES = [
      '安全杂项', '从此开始', '二进制漏洞审计', '密码学', '逆向工程', 'Web安全与渗透测试',
    ]

    const categoryTabs = computed(() => {
      const fromPlatform = platformCategories.value.length
        ? platformCategories.value
        : DEFAULT_CATEGORIES
      const cats = new Set(challenges.value.map(c => c.category).filter(Boolean))
      const merged = ['全部']
      for (const c of fromPlatform) {
        if (cats.has(c)) merged.push(c)
      }
      for (const c of cats) {
        if (!merged.includes(c)) merged.push(c)
      }
      return merged
    })
